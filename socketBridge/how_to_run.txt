using a test image 
python ultralytics_server.py --engine ~/drono/drono-ai/models/best1.engine --port 12345 --image test1.jpg

python ultralytics_server_camera.py --engine ~/drono/drono-ai/models/best1.engine --port 12345 --camera 0

main.cpp
SocketServer.cpp
SocketServer.h
